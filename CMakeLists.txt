cmake_minimum_required(VERSION 3.20)
project(ReachTerminal VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PostgreSQL REQUIRED)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(LibXml2 REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTPLIB REQUIRED IMPORTED_TARGET cpp-httplib>=0.18)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(REACH_ASSET_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(REACH_ASSETS frontend/index.html frontend/app.js frontend/release.js frontend/gateway-session.js frontend/style.css db/001_initial.sql)
add_custom_command(OUTPUT "${REACH_ASSET_DIR}/embedded_assets.cpp" "${REACH_ASSET_DIR}/embedded_assets.hpp"
 COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_assets.py" --root "${CMAKE_CURRENT_SOURCE_DIR}" --output "${REACH_ASSET_DIR}"
 DEPENDS scripts/embed_assets.py ${REACH_ASSETS} VERBATIM)
add_executable(reachterminal backend/main.cpp "${REACH_ASSET_DIR}/embedded_assets.cpp")
target_include_directories(reachterminal PRIVATE "${REACH_ASSET_DIR}")
target_compile_definitions(reachterminal PRIVATE REACH_ROOT="${CMAKE_SOURCE_DIR}")
target_link_libraries(reachterminal PRIVATE PostgreSQL::PostgreSQL CURL::libcurl OpenSSL::Crypto LibXml2::LibXml2 nlohmann_json::nlohmann_json PkgConfig::HTTPLIB Threads::Threads)
target_compile_options(reachterminal PRIVATE -Wall -Wextra -Wconversion)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto LibXml2::LibXml2 CURL::libcurl)
add_test(NAME core COMMAND core_tests)
add_executable(release_tests tests/release_tests.cpp)
target_link_libraries(release_tests PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto LibXml2::LibXml2 CURL::libcurl)
add_test(NAME release COMMAND release_tests)
option(REACH_BUILD_DESKTOP "Build the native GTK desktop frontend" ON)
if(REACH_BUILD_DESKTOP)
  set(REACH_PREV_PKG_SYSROOT "$ENV{PKG_CONFIG_SYSROOT_DIR}")
  set(REACH_PREV_PKG_PATH "$ENV{PKG_CONFIG_PATH}")
  set(ENV{PKG_CONFIG_SYSROOT_DIR} "")
  set(ENV{PKG_CONFIG_PATH} "")
  pkg_check_modules(GTK3 QUIET IMPORTED_TARGET gtk+-3.0>=3.24)
  set(ENV{PKG_CONFIG_SYSROOT_DIR} "${REACH_PREV_PKG_SYSROOT}")
  set(ENV{PKG_CONFIG_PATH} "${REACH_PREV_PKG_PATH}")
  if(GTK3_FOUND)
    add_executable(reachterminal-desktop desktop/main.cpp desktop/pages.cpp)
    target_link_libraries(reachterminal-desktop PRIVATE PkgConfig::GTK3 CURL::libcurl nlohmann_json::nlohmann_json Threads::Threads)
    target_compile_options(reachterminal-desktop PRIVATE -Wall -Wextra)
    add_executable(native_tests desktop/unit_tests.cpp)
    target_link_libraries(native_tests PRIVATE PkgConfig::GTK3 CURL::libcurl nlohmann_json::nlohmann_json Threads::Threads)
    add_test(NAME native-adapters COMMAND native_tests)
  else()
    message(STATUS "GTK 3.24 development files unavailable; install libgtk-3-dev for native frontend")
  endif()
endif()
